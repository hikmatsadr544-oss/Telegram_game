#include #include #include #include #include #include class ImperialCitadel { public: double gold = 0; int energy = 1000; int maxEnergy = 1000; int tapPower = 1; int tapCost = 100; int gps = 0; int mineCost = 50; std::atomic running{true}; void startEngine() { std::thread([this]() { while (running) { std::this_thread::sleep_for(std::chrono::seconds(1)); gold += gps; if (energy < maxEnergy) { energy = std::min(maxEnergy, energy + 1); } } }).detach(); } void tap() { if (energy >= tapPower) { gold += tapPower; energy -= tapPower; std::cout << "✨ کلیک کرا! +" << tapPower << " زێڕ | وزە: " << energy << "/" << maxEnergy << std::endl; } else { std::cout << "⚠️ وزەت تەواو بووە!" << std::endl; } } void buyFullTank() { if (gold >= 1000) { gold -= 1000; energy = maxEnergy; std::cout << "⚡ وزەکەت بە تەواوی پڕکرایەوە!" << std::endl; } else { std::cout << "❌ زێڕی پێویستت نییە!" << std::endl; } } void buyMultiTap() { if (gold >= tapCost) { gold -= tapCost; tapPower += 1; tapCost = std::floor(tapCost * 2); std::cout << "👆 هێزی کلیک بەرزکرایەوە بۆ +" << tapPower << std::endl; } else { std::cout << "❌ زێڕی پێویستت نییە!" << std::endl; } } void buyMine() { if (gold >= mineCost) { gold -= mineCost; gps += 1; mineCost = std::floor(mineCost * 1.5); std::cout << "⛏️ کانگای زێڕ کڕدرا! داهات: +" << gps << "/چرکە" << std::endl; } else { std::cout << "❌ زێڕی پێویستت نییە!" << std::endl; } } void claimDaily() { gold += 1000; std::cout << "🎁 پاداشتی ڕۆژانە (+1000 زێڕ) وەرگیرا!" << std::endl; } void tonShop(int opt) { if (opt == 1) { maxEnergy += 5000; energy += 5000; std::cout << "💎 +5,000 وزەی زیادە زێدە کرا!" << std::endl; } else if (opt == 2) { gold += 100000; std::cout << "💎 +100,000 زێڕی خێرا زێدە کرا!" << std::endl; } } void inviteFriend() { gold += 2000; std::cout << "👥 هاوڕێیەک بانگهێشت کرا! +2,000 زێڕ وەربگیرا." << std::endl; } void showStatus() { std::cout << "\n--- 🏰 Imperial Citadel Status ---" << std::endl; std::cout << "💰 زێڕ: " << (long)gold << " | ⚡ وزە: " << energy << "/" << maxEnergy << std::endl; std::cout << "📈 داهاتی چرکە: +" << gps << "/sec | 👆 هێزی کلیک: +" << tapPower << std::endl; std::cout << "-----------------------------------" << std::endl; } }; int main() { ImperialCitadel game; game.startEngine(); int choice = -1; while (choice != 0) { game.showStatus(); std::cout << "1: کلیک | 2: بووستەر | 3: دیاری ڕۆژانە | 4: فرۆشگای TON | 5: بانگهێشت | 0: دەرچوون\nهەڵبژاردن: "; std::cin >> choice; if (choice == 1) game.tap(); else if (choice == 2) { char sub; std::cout << "a: Full Tank | b: MultiTap | c: Mine\nکام بووستەر؟ "; std::cin >> sub; if (sub == 'a') game.buyFullTank(); else if (sub == 'b') game.buyMultiTap(); else if (sub == 'c') game.buyMine(); } else if (choice == 3) game.claimDaily(); else if (choice == 4) { int s; std::cout << "1: 0.1 TON (+5000 وزە) | 2: 0.5 TON (+100000 زێڕ)\nهەڵبژاردن: "; std::cin >> s; game.tonShop(s); } else if (choice == 5) game.inviteFriend(); } game.running = false; return 0; }
